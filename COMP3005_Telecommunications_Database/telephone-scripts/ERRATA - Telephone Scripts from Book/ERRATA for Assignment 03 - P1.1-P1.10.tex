\section{Problem 1.4: Universal Service Subscribers}
The requirement is to find subscribers who have every service listed in the system. This is a classic Relational Division problem.

\begin{raguery}{Relational Algebra}
    {\fontsize{12pt}{2pt}\selectfont $\pi_{name, address}(subscribers \bowtie (\gamma_{portid; count(service) \to c}(service\_subscribers) \ltimes \sigma_{c = (\gamma_{count(scode) \to tc}(services))}(services)))$}
\end{raguery}

\begin{lstlisting}
SELECT s.name, s.address
FROM subscribers s
JOIN service_subscribers ss ON s.portid = ss.portid
GROUP BY s.name, s.address
HAVING COUNT(DISTINCT ss.service) = (SELECT COUNT(*) FROM services);
\end{lstlisting}

\section{Problem 1.5: Service Threshold}
Find subscribers with at least three services.

\begin{raguery}{Relational Algebra}
    {\small $\pi_{name}(subscribers \bowtie \sigma_{count \ge 3}(\gamma_{portid; count(service) \to count}(service\_subscribers)))$}
\end{raguery}

\begin{lstlisting}
SELECT s.name
FROM subscribers s
JOIN service_subscribers ss ON s.portid = ss.portid
GROUP BY s.name
HAVING COUNT(ss.service) >= 3;
\end{lstlisting}

\section{Problem 1.6: Most Popular Service}
Identifies the service(s) with the maximum number of subscribers.

\begin{raguery}{Relational Algebra}
    Let $R = \gamma_{service; count(*) \to c}(service\_subscribers)$ \\
    Result: $\pi_{service}(\sigma_{c = max(c)}(R))$
\end{raguery}

\begin{lstlisting}
SELECT service FROM service_subscribers
GROUP BY service
HAVING COUNT(*) = (
    SELECT MAX(cnt) FROM (SELECT COUNT(*) AS cnt FROM service_subscribers GROUP BY service)
);
\end{lstlisting}

\section{Problem 1.7: Trunk Routing Preference}
Implements longest-prefix matching for area code 416 and office 334.

\begin{raguery}{Relational Algebra}
    {\fontsize{6.5pt}{2pt}\selectfont $\pi_{portid, foreign\_switch}(\sigma_{(area='416' \land office='334') \lor (area='416' \land office='000') \lor (area='000' \land office='000')}(trunks \bowtie trunk\_routes))$}
\end{raguery}

\begin{lstlisting}
SELECT t.portid, t.foreign_switch
FROM trunks t JOIN trunk_routes tr ON t.portid = tr.portid
WHERE (tr.area = '416' AND tr.office = '334') 
   OR (tr.area = '416' AND tr.office = '000') 
   OR (tr.area = '000' AND tr.office = '000')
ORDER BY CASE 
    WHEN tr.area = '416' AND tr.office = '334' THEN 1
    WHEN tr.area = '416' AND tr.office = '000' THEN 2
    ELSE 3 END ASC;
\end{lstlisting}

\section{Problem 1.8: Line Availability}
Checks the IDLE state for a specific directory number.

\begin{raguery}{Relational Algebra}
    $\pi_{portid, state}(\sigma_{area='613' \land office='712' \land number='0024'}(lines \bowtie channels))$
\end{raguery}

\begin{lstlisting}
SELECT l.portid, '(' || l.area || ') ' || l.office || '-' || l.number AS directory_number, c.state
FROM lines l JOIN channels c ON l.portid = c.portid
WHERE l.area = '613' AND l.office = '712' AND l.number = '0024';
\end{lstlisting}

\section{Problem 1.9: Routing with Availability}
Same as 1.7 but ensures at least one channel is 'IDLE'.

\begin{lstlisting}
SELECT DISTINCT t.portid, t.foreign_switch
FROM trunks t
JOIN trunk_routes tr ON t.portid = tr.portid
JOIN channels c ON t.portid = c.portid
WHERE (routing_criteria) AND c.state = 'IDLE'
ORDER BY preference_logic;
\end{lstlisting}

\section{Problem 1.10: Jason Allison Subset}
Finds subscribers who have at least all the services of Jason Allison (portid 2).

\begin{raguery}{Relational Algebra}
    $Subscribers \div (\pi_{service}(\sigma_{portid=2}(service\_subscribers)))$
\end{raguery}

\begin{lstlisting}
SELECT s.name FROM subscribers s
WHERE NOT EXISTS (
    SELECT ss1.service FROM service_subscribers ss1 WHERE ss1.portid = 2
    EXCEPT
    SELECT ss2.service FROM service_subscribers ss2 WHERE ss2.portid = s.portid
);
\end{lstlisting}
