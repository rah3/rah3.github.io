\section{Quiz 3 - Relational Algebra Review}
\textit{Questions based on the Telephone Switching Database Rev 4 schema.}

\begin{questions}
    \question[1] Which ONE of the following will list the $\text{call\_id}$'s for all calls that both originate and terminate in a foreign switch (where both originator and terminator are trunks)?
    \begin{choices}
        \choice $\pi_{\text{call\_id}}(\text{calls} * \text{trunks} * \text{trunks})$
        \choice $\pi_{\text{call\_id}}(\text{calls} \bowtie_{\text{orig}=\text{portid} \land \text{term}=\text{portid}} \text{trunks})$
        \choice $\pi_{\text{call\_id}} (\text{calls} \bowtie_{\text{orig}=\text{portid} \land \text{term}=\text{portid}} \text{trunks})$
        \choice $\sigma_{\text{call\_id}}(\text{calls} \bowtie_{\text{orig}=\text{term}} \text{trunks})$
        \CorrectChoice $\pi_{\text{call\_id}} (\pi_{\text{call\_id, term}}(\text{calls} \bowtie_{\text{orig}=\text{portid}} \text{trunks}) \bowtie_{\text{term}=\text{portid}} \text{trunks})$
    \end{choices}

    \question[1] Which ONE of the following would list the names of all the services and the number of lines that subscribe to that service?
    \begin{choices}
        \choice $\text{SERVICE\_COUNT}(\text{scode, num\_services}) = \newline
        (\text{service} \mathfrak{F}_{\text{count portid}}(\text{service\_subscribers})) \\ \pi_{\text{name, num\_services}}(\text{subscribers} * \text{SERVICE\_COUNT})$
        \CorrectChoice $\text{SERVICE\_COUNT}(\text{scode, num\_services}) = \newline
        (\text{service} \mathfrak{F}_{\text{count portid}}(\text{service\_subscribers})) \\ \pi_{\text{service, num\_services}}(\text{services} * \text{SERVICE\_COUNT})$
        \choice $\text{SERVICE\_COUNT}(\text{service, num\_services}) = \newline
        (\text{service} \mathfrak{F}_{\text{count portid}}(\text{service\_subscribers})) \\ \pi_{\text{service, num\_services}}(\text{services} * \text{SERVICE\_COUNT})$
        \choice $\pi_{\text{service, num\_services}} ((\text{service} \mathfrak{F}_{\text{count portid}}(\text{service\_subscribers})))$
    \end{choices}
    \begin{solution}
        Renaming the grouping attribute to scode is necessary because \text{service} in service\_subscribers is a foreign key, but \text{service} in the \text{services} table refers to the descriptive name. A natural join ($*$) requires identical column names to link the tables.
    \end{solution}

    \question[1] Which ONE of the following would list only the $\text{portid}$'s of the trunks that currently have at least one idle channel?
    \begin{choices}
        \choice $\pi_{\text{state}}(\sigma_{\text{state}=\text{`IDLE'}}(\text{channels} * \text{trunks}))$
        \CorrectChoice $\pi_{\text{portid}}(\text{trunks} * \pi_{\text{portid}}(\sigma_{\text{state}=\text{`IDLE'}}(\text{channels})))$
        \choice $\pi_{\text{portid}}(\sigma_{\text{state}=\text{`IDLE'}}(\text{trunks}))$
        \choice $\pi_{\text{portid}}(\sigma_{\text{state}=\text{`IDLE'}}(\text{channels}))$
        \choice $\pi_{\text{portid}}((\text{calls} \bowtie_{\text{orig}=\text{portid}} \text{trunks}) * \pi_{\text{portid}}(\sigma_{\text{state}=\text{`IDLE'}}(\text{channels})))$
    \end{choices}

    \question[1] Suppose a Line originator dialed (905) 238-1243. Which ONE of the following would find the $\text{portid}$'s of all the trunks that could route this call (including wildcards)?
    \begin{choices}
        \choice $\sigma_{\text{portid}}(\sigma_{\text{area=`905'} \lor \text{area=`000'}}(\text{trunks} * \text{channels}))$
        \choice $\pi_{\text{portid}}(\sigma_{\text{area=`905'} \lor \text{area=`000'}}(\text{trunk\_routes}))$
        \CorrectChoice $\pi_{\text{portid}}(\sigma_{(\text{area=`905'}
        \land \text{office=`238'}) \lor (\text{area='905'} \land \text{office=`000'}) \lor (\text{area=`000'} \land \text{office=`000'})}\newline (\text{trunk\_routes}))$
        \choice $\pi_{\text{portid}}(\sigma_{\text{state=`IDLE'}}(\text{trunk\_routes}))$
        \choice $\pi_{\text{portid}}(\sigma_{\text{foreign\_switch=`905,238'}}(\text{trunks}))$
    \end{choices}

    \question[1] Which ONE of these relational algebra queries will find the address of everyone that subscribes to the `Call Waiting' service ($\text{scode=`CWT'}$)?
    \begin{choices}
        \choice $\sigma_{\text{address}}(\sigma_{\text{service=`Call Waiting'}}(\text{service\_subscribers}))$
        \choice $\sigma_{\text{address}}(\text{subscribers} * \mathfrak{F}_{\text{COUNT scode}}(\text{services}))$
        \choice $\pi_{\text{address}}(\text{service\_subscribers} * \text{subscribers})$
        \choice $\pi_{\text{address}}(\sigma_{\text{service=`CWT'}}(\text{service\_subscribers} \bowtie_{\text{service='Call Waiting'}} \text{subscribers}))$
        \CorrectChoice $\pi_{\text{address}}(\sigma_{\text{service=`CWT'}}(\text{service\_subscribers} * \text{subscribers}))$
    \end{choices}
\end{questions}
